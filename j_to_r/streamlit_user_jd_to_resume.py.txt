# streamlit_user_jd_to_resume.py
# ----------------------------------------------------------------------
# JD → Resume semantic matcher (cosine only, no BM25)
# - NEW: Employers must provide an allow-list of resume_ids (applicants).
# - We compose JD text from a form, chunk/pool-embed via Ollama,
# - Then we ONLY score resumes whose IDs are in the allow-list.
# - We fetch all chunks for each allowed resume, compute cosine sims,
# - Keep best-chunk evidence per resume, and return relative Match%.
# ----------------------------------------------------------------------

def read_text_from_bytes(data: bytes, filename: str) -> str:
    """
    Extract text from common formats and clean it.
    Supports: .pdf, .docx, .txt, .rtf, .md, .doc (best-effort decode)
    """
    ext = Path(filename).suffix.lower()
    try:
        if ext == ".pdf":
            with io.BytesIO(data) as f:
                reader = PdfReader(f, strict=False)
                text = " ".join((page.extract_text() or "") for page in reader.pages)
        elif ext == ".docx":
            with io.BytesIO(data) as f:
                doc = Docx(f)
                text = "\n".join(p.text for p in doc.paragraphs)
        elif ext in {".txt", ".rtf", ".md"}:
            text = data.decode(errors="ignore")
        else:
            # .doc or unknown → best-effort decode
            text = data.decode(errors="ignore")
    except Exception:
        # Fallback to raw decode if any parser chokes
        text = data.decode(errors="ignore")
    return clean_text(text)


def _chunk_query_text(text: str, size: int, overlap: int, max_chunks: int) -> List[str]:
    """
    Split the (possibly long) JD text into smaller chunks so we can embed
    and pooled-average them for a more stable query vector.
    """
    t = text[:MAX_QUERY_CHARS] if MAX_QUERY_CHARS > 0 else text
    n = len(t)
    if n == 0:
        return []
    size = max(1, int(size))
    overlap = max(0, int(overlap))
    if overlap >= size:
        overlap = size // 4  # safety
    chunks: List[str] = []
    start = 0
    while start < n and len(chunks) < max_chunks:
        end = min(n, start + size)
        chunks.append(t[start:end])
        if end == n:
            break
        start = end - overlap
    return chunks


def _embed_query_ollama(chunk: str) -> List[float]:
    """
    Call Ollama's embeddings API directly.
    Requires: ollama serve  &  ollama pull <EMBED_MODEL>
    """
    resp = requests.post(
        f"{OLLAMA_HOST}/api/embeddings",
        json={"model": EMBED_MODEL, "prompt": chunk},
        timeout=OLLAMA_TIMEOUT,
    )
    resp.raise_for_status()
    data = resp.json()
    # Ollama returns either {'embedding': [...]} or {'data': [{'embedding': [...]}]}
    vec = data.get("embedding") or (data.get("data") or [{}])[0].get("embedding")
    if not isinstance(vec, list) or not all(isinstance(x, (int, float)) for x in vec):
        raise ValueError("Invalid embedding vector from Ollama")
    return [float(x) for x in vec]


def _embed_query_pooled(text: str) -> List[float]:
    """
    Pool embeddings across several overlapping chunks of the JD text.
    Length-weighted mean pooling.
    """
    chunks = _chunk_query_text(text, R2J_QUERY_CHUNK_SIZE, R2J_QUERY_CHUNK_OVERLAP, R2J_QUERY_MAX_CHUNKS)
    if not chunks:
        return []
    vecs = [_embed_query_ollama(ch) for ch in chunks]
    dims = len(vecs[0])
    pooled = [0.0] * dims
    weights = [len(ch) for ch in chunks]
    denom = float(sum(weights) or 1.0)
    for v, w in zip(vecs, weights):
        if len(v) != dims:
            # Skip malformed vectors (defensive)
            continue
        wf = float(w)
        for i in range(dims):
            pooled[i] += wf * float(v[i])
    for i in range(dims):
        pooled[i] /= denom
    return pooled


def _normalize(xs: List[float]) -> List[float]:
    """Min–max normalize to [0,1] (stable even with equal values)."""
    if not xs:
        return xs
    lo, hi = min(xs), max(xs)
    if hi - lo < 1e-12:
        return [0.0 for _ in xs]
    scale = hi - lo
    return [(x - lo) / scale for x in xs]


def _get_resume_collection():
    client = chromadb.PersistentClient(path=CHROMA_DIR_RESUMES)
    return client.get_or_create_collection(name=CHROMA_COLLECTION_RESUMES)


def _safe_cosine(a: List[float], b: List[float], eps: float = 1e-12) -> float:
    """
    Cosine similarity mapped to [0,1]. Returns 0 if either vector is near-zero.
    """
    va = np.asarray(a, dtype=float)
    vb = np.asarray(b, dtype=float)
    na = float(np.linalg.norm(va))
    nb = float(np.linalg.norm(vb))
    if na < eps or nb < eps:
        return 0.0
    cos = float(np.dot(va, vb) / (na * nb))
    cos = max(-1.0, min(1.0, cos))
    return 0.5 * (cos + 1.0)  # [-1,1] → [0,1]


def _get_all_parent_embeddings(coll, parent_id: str) -> Tuple[List[List[float]], List[str], List[Dict]]:
    """
    Fetch ALL chunk embeddings for a given resume (parent_id).
    We rely on metadata 'resume_id' being set at ingestion.
    """
    all_embs: List[List[float]] = []
    all_docs: List[str] = []
    all_meta: List[Dict] = []
    offset = 0
    page = 500
    while True:
        batch = coll.get(
            where={"resume_id": parent_id},
            include=["embeddings", "documents", "metadatas"],
            limit=page,
            offset=offset,
        )
        ids = batch.get("ids", [])
        if not ids:
            break
        embs = batch.get("embeddings", [])
        docs = batch.get("documents", [])
        meta = batch.get("metadatas", [])
        all_embs.extend(embs or [])
        all_docs.extend(docs or [])
        all_meta.extend(meta or [])
        offset += len(ids)
    return all_embs, all_docs, all_meta


def _parse_allowed_ids(raw: str) -> List[str]:
    """
    Parse a comma/space/newline-separated list of resume_ids.
    """
    if not raw:
        return []
    # split on commas or whitespace, strip empty tokens
    parts = [p.strip() for chunk in raw.split(",") for p in chunk.split()]
    return [p for p in parts if p]


def _semantic_search_guarded(jd_text: str, coll, allowed_resume_ids: List[str]) -> List[Dict]:
    """
    Guarded search: ONLY compute scores for resume_ids in allowed_resume_ids.
    For each allowed resume: fetch all its chunk embeddings, compute cosine sims
    vs the pooled JD vector, and keep best-chunk evidence.
    """
    qvec = _embed_query_pooled(jd_text)
    if not qvec:
        return []

    rows: List[Dict] = []
    sims_all: List[float] = []

    for rid in allowed_resume_ids:
        embs, docs, metas = _get_all_parent_embeddings(coll, rid)
        if not embs:
            continue

        # compute sims for each chunk and find the best
        sims = [_safe_cosine(qvec, e) for e in embs]
        best_idx = int(np.argmax(sims))
        best_sim = float(sims[best_idx])
        sims_all.append(best_sim)

        md0 = metas[0] if metas else {}
        best_doc = (docs[best_idx] if 0 <= best_idx < len(docs) else "") or ""
        preview = best_doc[:320].replace("\n", " ")

        rows.append({
            "resume_id": rid,
            "candidate_name": (md0 or {}).get("candidate_name"),
            "email": (md0 or {}).get("email"),
            "phone": (md0 or {}).get("phone"),
            "sim": best_sim,
            "preview": preview,
        })

    if not rows:
        return []

    # Relative Match% scaling within this guarded result set
    hi = max(sims_all) or 1.0
    lo = min(sims_all)
    span = hi - lo if hi > lo else 1.0
    for r in rows:
        r["match_pct"] = round(100.0 * ((r["sim"] - lo) / span), 1)

    # rank by sim desc and keep top-K
    rows = sorted(rows, key=lambda x: x["sim"], reverse=True)[: R2J_TOP_K_FINAL]
    return rows
